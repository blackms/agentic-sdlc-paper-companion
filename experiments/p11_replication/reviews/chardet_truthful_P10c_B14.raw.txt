class CharDistributionAnalysis(object):
    ENOUGH_DATA_THRESHOLD = 1024
    SURE_YES = 0.99
    SURE_NO = 0.01
    MINIMUM_DATA_THRESHOLD = 3

    def __init__(self):
        # Mapping table to get frequency order from char order (get from
        # GetOrder())
        self._char_to_freq_order = None
        self._table_size = None  # Size of above table
        # This is a constant value which varies from language to language,
        # used in calculating confidence.  See
        # http://www.mozilla.org/projects/intl/UniversalCharsetDetection.html
        # for further detail.
        self.typical_distribution_ratio = None
        self._done = None
        self._total_chars = None
        self._freq_chars = None
        self.reset()

    def reset(self):
        """reset analyser, clear any state"""
        # If this flag is set to True, detection is done and conclusion has
        # been made
        self._done = False
        self._total_chars = 0  # Total characters encountered
        # The number of characters whose frequency order is less than 512
        self._freq_chars = 0

    def feed(self, char, char_len):
        """feed a character with known length"""
        if char_len == 2:
            # we only care about 2-bytes character in our distribution analysis
            order = self.get_order(char)
        else:
            order = -1
        if order >= 0:
            self._total_chars += 1
            # order is valid
            if order < self._table_size:
                if 512 > self._char_to_freq_order[order]:
                    self._freq_chars += 1

    def get_confidence(self):
        """return confidence based on existing data"""
        # if we didn't receive any character in our consideration range,
        # return negative answer
        if self._total_chars <= 0 or self._freq_chars <= self.MINIMUM_DATA_THRESHOLD:
            return self.SURE_NO

        if self._total_chars != self._freq_chars:
            r = (self._freq_chars / ((self._total_chars - self._freq_chars)
                 * self.typical_distribution_ratio))
            if r < self.SURE_YES:
                return r

        # normalize confidence (we don't want to be 100% sure)
        return self.SURE_YES

    def got_enough_data(self):
        # It is not necessary to receive all data to draw conclusion.
        # For charset detection, certain amount of data is enough
        return self._total_chars > self.ENOUGH_DATA_THRESHOLD

    def get_order(self, byte_str):
        # We do not handle characters based on the original encoding string,
        # but convert this encoding string to a number, here called order.
        # This allows multiple encodings of a language to share one frequency
        # table.
        return -1


class EUCTWDistributionAnalysis(CharDistributionAnalysis):

# Contracts: chardet_dom

## Module overview

This module implements character distribution analyzers used by charset detection for CJK encodings. It converts validated 2-byte character sequences into language-neutral frequency-table orders, counts how many observed characters are common, and derives a confidence score from those counts.

There are no top-level functions. The module exposes a base `CharDistributionAnalysis` class and encoding-specific subclasses for EUC-TW, EUC-KR, GB2312, Big5, Shift-JIS, and EUC-JP.

## Function/method contracts

- **Signature**
  ```python
  def __init__(self)
  ```

- **Pre-conditions**
  - Called with no arguments.
  - Subclasses may rely on this initializer to create shared state before installing encoding-specific tables.
  - No frequency table is required for the base class at construction time.

- **Post-conditions**
  - `_char_to_freq_order is None`.
  - `_table_size is None`.
  - `typical_distribution_ratio is None`.
  - `_done is False`.
  - `_total_chars == 0`.
  - `_freq_chars == 0`.
  - `reset()` has been invoked exactly to initialize analyzer counters.

- **Invariants**
  - After initialization, the instance is in the same counting state as after `reset()`.
  - The base class does not choose an encoding table.

- **Edge cases**
  - Direct construction of `CharDistributionAnalysis` is allowed, but useful frequency analysis is impossible because no table or ratio is installed.
  - Calling `feed()` on the base class remains safe because base `get_order()` always returns `-1`.

### `CharDistributionAnalysis.feed`

- **Signature**
  ```python
  def feed(self, char, char_len)
  ```

- **Pre-conditions**
  - `char_len` is an integer-like value representing the known byte length of `char`.
  - If `char_len == 2`, `char` must be indexable at positions `0` and `1`.
  - For concrete subclasses, `char[0]` and `char[1]` must be integers in byte range `0..255`, as produced by indexing `bytes`/`bytearray` in Python 3.
  - For concrete subclasses, `_char_to_freq_order`, `_table_size`, and `typical_distribution_ratio` must have been initialized by the subclass constructor.
  - Upstream encoding state machines are expected to have validated byte-sequence legality; this method does not fully validate byte ranges.

- **Post-conditions**
  - If `char_len != 2`, no counters change.
  - If `char_len == 2`, `get_order(char)` is called once.
  - If `order < 0`, no counters change.
  - If `order >= 0`, `_total_chars` increases by exactly `1`.
  - If `0 <= order < _table_size` and `_char_to_freq_order[order] < 512`, `_freq_chars` increases by exactly `1`.
  - If `order >= _table_size`, `_total_chars` still increases but `_freq_chars` does not.
  - `_done` is not changed.

- **Invariants**
  - `_total_chars >= 0`.
  - `_freq_chars >= 0`.
  - `_freq_chars <= _total_chars`.
  - Feeding data never decreases counters.
  - Only valid non-negative orders contribute to `_total_chars`.
  - Only orders mapped to frequency ranks strictly less than `512` contribute to `_freq_chars`.

- **Edge cases**
  - `char_len == 2` with a `char` shorter than 2 bytes may raise `IndexError` from `get_order()`.
  - Non-integer byte values may raise `TypeError` during comparisons/arithmetic.
  - `char_len` is trusted; the method does not compare it to `len(char)`.
  - Base-class `feed()` with `char_len == 2` does not count anything because base `get_order()` returns `-1`.

- **Operator-level invariants**
  - The frequent-character test is strict: `_char_to_freq_order[order] < 512`, implemented as `512 > value`.
  - The table-bound test is strict upper-bound: `order < _table_size`.
  - `_total_chars` increments before frequency-table lookup whenever `order >= 0`.

### `CharDistributionAnalysis.get_confidence`

- **Signature**
  ```python
  def get_confidence(self)
  ```

- **Pre-conditions**
  - The instance has been constructed.
  - For meaningful non-minimum confidence, `typical_distribution_ratio` must be a positive numeric value.
  - `_total_chars` and `_freq_chars` are non-negative integers with `_freq_chars <= _total_chars`.

- **Post-conditions**
  - Returns a numeric confidence value.
  - If `_total_chars <= 0`, returns `SURE_NO` (`0.01`).
  - If `_freq_chars <= MINIMUM_DATA_THRESHOLD` (`3`), returns `SURE_NO` (`0.01`).
  - Otherwise, if `_total_chars != _freq_chars`, computes:
    ```python
    r = _freq_chars / (
        (_total_chars - _freq_chars) * typical_distribution_ratio
    )
    ```
    - If `r < SURE_YES` (`0.99`), returns `r`.
    - If `r >= SURE_YES`, returns `SURE_YES`.
  - If `_total_chars == _freq_chars` and minimum thresholds are satisfied, returns `SURE_YES` (`0.99`).
  - Does not mutate analyzer state.

- **Invariants**
  - Return value is capped above by `0.99`.
  - Return value is never `1.0`.
  - Return value is exactly `0.01` when there is no usable data or too few frequent characters.
  - Confidence depends only on `_total_chars`, `_freq_chars`, and `typical_distribution_ratio`.

- **Edge cases**
  - `_freq_chars == 3` returns `0.01`; at least `4` frequent characters are required for computed confidence.
  - `_total_chars == _freq_chars` avoids division by zero and returns `0.99`.
  - If `typical_distribution_ratio <= 0`, behavior is outside contract and may raise or produce invalid confidence.
  - If state invariants are broken externally, e.g. `_freq_chars > _total_chars`, the formula may return nonsensical values.

- **Operator-level invariants**
  - Minimum-data comparison is inclusive: `_freq_chars <= 3` returns `SURE_NO`.
  - Confidence cap comparison is strict: only `r < 0.99` returns `r`; `r == 0.99` returns `0.99`.

## Module overview

This module implements character distribution analyzers used by charset detection for CJK encodings. It converts validated 2-byte character sequences into language-neutral frequency-table orders, counts how many observed characters are common, and derives a confidence score from those counts.

There are no top-level functions. The module exposes a base `CharDistributionAnalysis` class and encoding-specific subclasses for EUC-TW, EUC-KR, GB2312, Big5, Shift-JIS, and EUC-JP.

### `CharDistributionAnalysis.get_confidence`

- **Signature**
  ```python
  def get_confidence(self)
  ```

- **Pre-conditions**
  - The instance has been constructed.
  - For meaningful non-minimum confidence, `typical_distribution_ratio` must be a positive numeric value.
  - `_total_chars` and `_freq_chars` are non-negative integers with `_freq_chars <= _total_chars`.

- **Post-conditions**
  - Returns a numeric confidence value.
  - If `_total_chars <= 0`, returns `SURE_NO` (`0.01`).

    ```python
    r = _freq_chars / (
        (_total_chars - _freq_chars) * typical_distribution_ratio
    )
    ```
    - If `r < SURE_YES` (`0.99`), returns `r`.
    - If `r >= SURE_YES`, returns `SURE_YES`.
  - If `_total_chars == _freq_chars` and minimum thresholds are satisfied, returns `SURE_YES` (`0.99`).
  - Does not mutate analyzer state.

- **Invariants**
  - Return value is capped above by `0.99`.
  - Return value is never `1.0`.
  - Return value is exactly `0.01` when there is no usable data or too few frequent characters.
  - Confidence depends only on `_total_chars`, `_freq_chars`, and `typical_distribution_ratio`.

- **Edge cases**
  - `_freq_chars == 3` returns `0.01`; at least `4` frequent characters are required for computed confidence.
  - `_total_chars == _freq_chars` avoids division by zero and returns `0.99`.
  - If `typical_distribution_ratio <= 0`, behavior is outside contract and may raise or produce invalid confidence.
  - If state invariants are broken externally, e.g. `_freq_chars > _total_chars`, the formula may return nonsensical values.

- **Operator-level invariants**
  - Minimum-data comparison is inclusive: `_freq_chars <= 3` returns `SURE_NO`.
  - Confidence cap comparison is strict: only `r < 0.99` returns `r`; `r == 0.99` returns `0.99`.

class CharDistributionAnalysis(object):
    ENOUGH_DATA_THRESHOLD = 1024
    SURE_YES = 0.99
    SURE_NO = 0.01
    MINIMUM_DATA_THRESHOLD = 3

    def __init__(self):
        self._char_to_freq_order = None
        self._table_size = None
        self.typical_distribution_ratio = None
        self._done = None
        self._total_chars = None
        self._freq_chars = None
        self.reset()

    def reset(self):
        """reset analyser, clear any state"""
        self._done = False
        self._total_chars = 0
        self._freq_chars = 0

    def feed(self, char, char_len):
        """feed a character with known length"""
        if char_len == 2:
            order = self.get_order(char)
        else:
            order = -1
        if order >= 0:
            self._total_chars += 1
            if order < self._table_size:
                if 512 > self._char_to_freq_order[order]:
                    self._freq_chars += 1

    def get_confidence(self):
        """return confidence based on existing data"""
        if self._total_chars <= 0 or self._freq_chars <= self.MINIMUM_DATA_THRESHOLD:
            return self.SURE_NO
        if self._total_chars != self._freq_chars:
            r = self._freq_chars / ((self._total_chars - self._freq_chars) * self.typical_distribution_ratio)
            if r < self.SURE_YES:
                return r
        return self.SURE_YES

    def got_enough_data(self):
        return self._total_chars > self.ENOUGH_DATA_THRESHOLD

    def get_order(self, byte_str):
        return -1
