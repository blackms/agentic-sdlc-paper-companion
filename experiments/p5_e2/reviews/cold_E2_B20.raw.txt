# Contratti formali (per cold contract-first reviewer)

Questi sono i **contratti dichiarati** delle 5 funzioni in `finance_lib.py`. Il cold reviewer riceve SOLO questo documento, NON il codice. Sulla base dei contratti deve identificare violazioni potenziali.

---

## `compound_interest(principal, annual_rate, years, compounds_per_year=1) -> Decimal`

**Inputs**:
- `principal` ≥ 0 (Decimal)
- `annual_rate` ≥ 0 (Decimal)
- `years` ≥ 0 (int)
- `compounds_per_year` ≥ 1 (int)

**Output**: `M = principal · (1 + annual_rate / compounds_per_year) ^ (compounds_per_year · years)`

**Invarianti**:
- I1: `M ≥ principal` ∀ inputs validi (monotonicità in t)
- I2: `M = principal` se `years = 0` o `principal = 0` o `annual_rate = 0`
- I3: monotonicità in r e t
- I4: determinismo: stesso input → stesso output
- I5: precisione: NO float per money, solo Decimal

**Errori**: `ValueError` se rate < 0, years < 0, compounds_per_year < 1.

---

## `loan_payment(principal, monthly_rate, months) -> Decimal`

**Inputs**:
- `principal` ≥ 0 (Decimal)
- `monthly_rate` ≥ 0 (Decimal)
- `months` > 0 (int)

**Output (formula amortizzazione)**:
- se `monthly_rate == 0`: `principal / months`
- altrimenti: `principal · r · (1+r)^months / ((1+r)^months − 1)`

**Invarianti**:
- I1: `payment > 0` per tutti gli input validi con principal > 0
- I2: `payment · months ≥ principal` (rimborso totale ≥ capitale)
- I3: continuità a `r → 0`: il limite della formula generale = `principal/months`
- I4: NO float per money

**Errori**: `ValueError` se principal < 0, rate < 0, months ≤ 0.

---

## `transfer(account_from, account_to, amount) -> Receipt`

**Inputs**:
- `account_from`, `account_to`: `Account` con `balance: Decimal`, `currency: str`, `id: str`
- `amount`: Decimal > 0

**Output**: `Receipt(from_id, to_id, amount)` se successo.

**Invarianti**:
- I1: **conservazione**: `from.balance + to.balance` prima = dopo
- I2: **atomicità**: se qualsiasi validazione/operazione fallisce, NESSUN balance modificato
- I3: stesso currency obbligatorio
- I4: no self-transfer
- I5: balance ≥ 0 sempre dopo l'operazione (no overdraft)
- I6: NO float

**Errori**: `ValueError` se amount ≤ 0, currency mismatch, self-transfer, balance < amount.

---

## `apply_fee(amount, fee_pct, fee_min, fee_max) -> Decimal`

**Inputs**:
- `amount` ≥ 0 (Decimal)
- `fee_pct` ≥ 0 (Decimal)
- `fee_min` ≥ 0 (Decimal)
- `fee_max` ≥ 0 (Decimal)
- `fee_min ≤ fee_max`

**Output**: `amount − clip(amount · fee_pct, fee_min, fee_max)`

**Invarianti**:
- I1: il risultato è ≤ amount (la fee viene SOTTRATTA)
- I2: la fee effettiva è in [fee_min, fee_max]
- I3: NO float

**Errori**: `ValueError` se uno qualunque degli input è negativo o se `fee_min > fee_max`.

---

## `fx_convert(amount, from_currency, to_currency, rates) -> Decimal`

**Inputs**:
- `amount` ≥ 0 (Decimal)
- `from_currency`, `to_currency`: str (codici ISO 4217)
- `rates`: dict `{(from, to): Decimal}` lookup table

**Output**:
- Se `from == to`: `amount` (identità)
- Altrimenti: `amount · rates[(from, to)]`

**Invarianti**:
- I1: identità per stessa currency
- I2: NO float per rate o amount
- I3: errore esplicito se rate mancante (no silent fallback)

**Errori**: `ValueError` se amount < 0, rate non disponibile.

class Reporter:
    """Print structured status lines and accumulate worst exit code."""

    def __init__(self) -> None:
        self.exit_code = EXIT_OK

    def ok(self, msg: str) -> None:
        print(f"[OK]   {msg}")

    def warn(self, msg: str) -> None:
        print(f"[WARN] {msg}")

    def fail(self, msg: str) -> None:
        print(f"[FAIL] {msg}")
        if self.exit_code < EXIT_FAIL:
            self.exit_code = EXIT_FAIL

    def config_error(self, msg: str) -> None:
        print(f"[CFG]  {msg}", file=sys.stderr)
        if self.exit_code < EXIT_CONFIG_ERROR:
            self.exit_code = EXIT_CONFIG_ERROR

def load_yaml_file(path: Path) -> dict:
    with path.open("r", encoding="utf-8") as fh:
        return yaml.safe_load(fh)


def load_json_file(path: Path) -> dict:
    with path.open("r", encoding="utf-8") as fh:
        return json.load(fh)


_profile_cache = {}
def load_profile(repo_root: Path, override: Path | None = None) -> dict | None:
    path = override if override is not None else repo_root / ".banking-profile"
    if not path.is_file():
        return None
    return load_yaml_file(path)


def file_matches_globs(rel_path: str, patterns: Iterable[str]) -> bool:
    return any(fnmatch.fnmatch(rel_path, pat) for pat in patterns)


def function_is_exempted(rel_file: str, func_name: str, exemptions: list[dict]) -> tuple[bool, str]:
    for ex in exemptions or []:
        if fnmatch.fnmatch(rel_file, ex["path"]) and fnmatch.fnmatch(func_name, ex["function_pattern"]):
            return True, ex.get("reason", "")
    return False, ""


def iter_files_under_sensitive(repo_root: Path, sensitive_paths: list[str]) -> Iterable[Path]:
    """Yield every file matching at least one sensitive_paths glob, relative to repo_root."""
    seen: set[Path] = set()
    for pattern in sensitive_paths:
        # fnmatch globs like "src/portfolio/**" => use Path.rglob
        # Strip trailing /** for rglob; otherwise treat as simple glob
        if pattern.endswith("/**"):
            base = repo_root / pattern[:-3]
            if base.is_dir():
                for p in base.rglob("*"):
                    if p.is_file() and p not in seen:
                        seen.add(p)
                        yield p
        else:
            for p in repo_root.glob(pattern):
                if p.is_file() and p not in seen:
                    seen.add(p)
                    yield p

def cmd_profile_validate(args: argparse.Namespace) -> int:
    reporter = Reporter()
    repo_root = Path(args.repo_root).resolve()
    profile_path = (
        Path(args.profile).resolve() if args.profile else repo_root / ".banking-profile"
    )

    if not profile_path.is_file():
        reporter.config_error(f"profile not found at {profile_path}")
        return reporter.exit_code

    try:
        profile = load_yaml_file(profile_path)
    except yaml.YAMLError as exc:
        reporter.config_error(f"profile is not valid YAML: {exc}")
        return reporter.exit_code

    try:
        schema = load_json_file(PROFILE_SCHEMA_PATH)
    except (FileNotFoundError, json.JSONDecodeError) as exc:
        reporter.config_error(f"cannot load schema {PROFILE_SCHEMA_PATH}: {exc}")
        return reporter.exit_code

    validator = jsonschema.Draft202012Validator(schema)
    errors = list(validator.iter_errors(profile))
    if not errors:
        reporter.ok(f"profile {profile_path.name} validates against schema")
        return reporter.exit_code

    for err in errors:
        path = "/".join(str(p) for p in err.absolute_path) or "<root>"
        reporter.fail(f"profile {profile_path.name} @ {path}: {err.message}")
    return reporter.exit_code

def _branch_dir(repo_root: Path, branch: str) -> Path:
    """Map a branch name to its on-disk directory under .claude/reviews/."""
    safe = branch.replace("/", "__")
    return repo_root / ".claude" / "reviews" / safe


def cmd_review_validate(args: argparse.Namespace) -> int:
    reporter = Reporter()
    repo_root = Path(args.repo_root).resolve()
    branch_dir = _branch_dir(repo_root, args.branch)

    if not branch_dir.is_dir():
        # Fall back to literal name (already filesystem-safe or example fixture).
        alt = repo_root / ".claude" / "reviews" / args.branch
        if alt.is_dir():
            branch_dir = alt
        else:
            reporter.config_error(f"review directory not found: {branch_dir}")
            return reporter.exit_code

    manifest_schema = load_json_file(MANIFEST_SCHEMA_PATH)
    findings_schema = load_json_file(FINDINGS_SCHEMA_PATH)
    mv = jsonschema.Draft202012Validator(manifest_schema)
    fv = jsonschema.Draft202012Validator(findings_schema)

    files = sorted(branch_dir.glob("*.json"))
    if not files:
        reporter.fail(f"no JSON files in {branch_dir}")
        return reporter.exit_code

    for fp in files:
        try:
            doc = load_json_file(fp)
        except json.JSONDecodeError as exc:
            reporter.fail(f"{fp.name}: invalid JSON: {exc}")
            continue
        if fp.name == "manifest.json":
            errors = list(mv.iter_errors(doc))
        else:
            errors = list(fv.iter_errors(doc))
        if not errors:
            reporter.ok(f"{fp.name} validates")
        else:
            for err in errors:
                path = "/".join(str(p) for p in err.absolute_path) or "<root>"
                reporter.fail(f"{fp.name} @ {path}: {err.message}")
    return reporter.exit_code

def _extract_contracts_from_text(text: str) -> list[tuple[dict | None, str | None]]:
    """Extract every ---contract block. Return list of (parsed_dict, parse_error_or_None)."""
    out: list[tuple[dict | None, str | None]] = []
    for m in CONTRACT_BLOCK_RE.finditer(text):
        body = m.group("body")
        try:
            parsed = yaml.safe_load(body)
            if not isinstance(parsed, dict):
                out.append((None, "contract block did not parse to a mapping"))
            else:
                out.append((parsed, None))
        except yaml.YAMLError as exc:
            out.append((None, str(exc)))
    return out


def cmd_contract_presence(args: argparse.Namespace) -> int:
    reporter = Reporter()
    repo_root = Path(args.repo_root).resolve()
    profile = load_profile(repo_root, Path(args.profile).resolve() if args.profile else None)
    if profile is None:
        reporter.config_error(".banking-profile not found; nothing to check")
        return reporter.exit_code

    sensitive_paths = profile.get("sensitive_paths", [])
    exemptions = profile.get("contract_exemptions", []) or []
    unsupported_languages: dict[str, int] = {}

    for fp in iter_files_under_sensitive(repo_root, sensitive_paths):
        rel = fp.relative_to(repo_root).as_posix()
        suffix = fp.suffix
        if suffix != ".py":
            unsupported_languages[suffix] = unsupported_languages.get(suffix, 0) + 1
            continue
        try:
            text = fp.read_text(encoding="utf-8")
        except (OSError, UnicodeDecodeError) as exc:
            reporter.fail(f"{rel}: cannot read ({exc})")
            continue

        contracts = _extract_contracts_from_text(text)
        contract_names = {
            c.get("name") for c, _ in contracts if isinstance(c, dict) and c.get("name")
        }
        funcs = [m.group("name") for m in PYTHON_DEF_RE.finditer(text)]
        # Skip dunders by convention: __init__ etc. are usually structural; explicit
        # exemption is still required to silence the check entirely.
        for func in funcs:
            exempted, reason = function_is_exempted(rel, func, exemptions)
            if exempted:
                reporter.ok(f"{rel}::{func} exempted ({reason})")
                continue
            if func in contract_names:
                reporter.ok(f"{rel}::{func} has contract")
            else:
                reporter.fail(f"{rel}::{func} is missing a ---contract block")

    for ext, count in sorted(unsupported_languages.items()):
        reporter.warn(
            f"{count} file(s) with extension {ext or '<none>'} in sensitive paths skipped — "
            "contract-presence V1 supports only Python; verify manually."
        )
    return reporter.exit_code

def cmd_contract_validate(args: argparse.Namespace) -> int:
    reporter = Reporter()
    repo_root = Path(args.repo_root).resolve()
    profile = load_profile(repo_root, Path(args.profile).resolve() if args.profile else None)
    if profile is None:
        reporter.config_error(".banking-profile not found; nothing to check")
        return reporter.exit_code

    schema = load_json_file(CONTRACT_SCHEMA_PATH)
    validator = jsonschema.Draft202012Validator(schema)

    sensitive_paths = profile.get("sensitive_paths", [])
    total = 0
    for fp in iter_files_under_sensitive(repo_root, sensitive_paths):
        try:
            text = fp.read_text(encoding="utf-8")
        except (OSError, UnicodeDecodeError):
            continue
        rel = fp.relative_to(repo_root).as_posix()
        for parsed, err in _extract_contracts_from_text(text):
            total += 1
            if err is not None:
                reporter.fail(f"{rel}: contract block parse error: {err}")
                continue
            errors = list(validator.iter_errors(parsed))
            label = parsed.get("name", "<unnamed>") if isinstance(parsed, dict) else "<invalid>"
            if not errors:
                reporter.ok(f"{rel}::{label} contract validates")
            else:
                for e in errors:
                    p = "/".join(str(x) for x in e.absolute_path) or "<root>"
                    reporter.fail(f"{rel}::{label} @ {p}: {e.message}")
    if total == 0:
        reporter.warn("no ---contract blocks found in any sensitive-path file")
    return reporter.exit_code

def cmd_floats_check(args: argparse.Namespace) -> int:
    reporter = Reporter()
    repo_root = Path(args.repo_root).resolve()
    profile = load_profile(repo_root, Path(args.profile).resolve() if args.profile else None)
    if profile is None:
        reporter.config_error(".banking-profile not found; nothing to check")
        return reporter.exit_code

    sensitive_paths = profile.get("sensitive_paths", [])
    hits = 0
    files_checked = 0
    skipped: dict[str, int] = {}

    for fp in iter_files_under_sensitive(repo_root, sensitive_paths):
        suffix = fp.suffix
        patterns = FLOAT_PATTERNS.get(suffix)
        if patterns is None:
            skipped[suffix] = skipped.get(suffix, 0) + 1
            continue
        files_checked += 1
        try:
            text = fp.read_text(encoding="utf-8")
        except (OSError, UnicodeDecodeError):
            continue
        rel = fp.relative_to(repo_root).as_posix()
        for pattern in patterns:
            regex = re.compile(pattern)
            for lineno, line in enumerate(text.splitlines(), start=1):
                if regex.search(line):
                    reporter.fail(f"{rel}:{lineno} float pattern '{pattern}': {line.strip()[:120]}")
                    hits += 1

    if hits == 0 and files_checked > 0:
        reporter.ok(f"no float patterns in {files_checked} sensitive-path file(s)")
    elif files_checked == 0:
        reporter.warn("no files with known extensions found in sensitive paths")
    for ext, count in sorted(skipped.items()):
        reporter.warn(f"{count} file(s) with extension {ext or '<none>'} skipped — floats-check has no pattern set for this extension")
    return reporter.exit_code

def _latest_round_per_agent(branch_dir: Path) -> dict[str, dict]:
    out: dict[str, dict] = {}
    for fp in sorted(branch_dir.glob("round-*.json")):
        try:
            doc = load_json_file(fp)
        except json.JSONDecodeError:
            continue
        agent = doc.get("agent")
        if not agent:
            continue
        prev = out.get(agent)
        if prev is None or doc.get("round", 0) > prev.get("round", 0):
            out[agent] = doc
    return out


def cmd_review_gate(args: argparse.Namespace) -> int:
    """Enforce the merge gate per DUAL-AGENT-REVIEW.md.

    The protocol caps the loop at 3 rounds and gives round 3 (Code Reviewer
    response) the closing word. Math Auditor's last verdict typically remains
    request-changes from round 2 even after fixes; the gate therefore looks
    at the LATEST round across all agents to determine resolution status,
    not at each agent's verdict in isolation.

    Pass conditions:
    - manifest.json present.
    - Both agents have at least one round file (so contract-first audit happened).
    - The latest round file overall has verdict=approve.
    - The latest round file by code-reviewer has verdict=approve.
    - Math Auditor produced a non-empty contract_only_summary in its round 2.
    - Every finding (deduped by id, taking the most recent ack) of severity
      blocker or major has ack.stance=resolved.
    """
    reporter = Reporter()
    repo_root = Path(args.repo_root).resolve()
    branch_dir = _branch_dir(repo_root, args.branch)
    if not branch_dir.is_dir():
        alt = repo_root / ".claude" / "reviews" / args.branch
        if alt.is_dir():
            branch_dir = alt
        else:
            reporter.config_error(f"review directory not found: {branch_dir}")
            return reporter.exit_code

    manifest_path = branch_dir / "manifest.json"
    if not manifest_path.is_file():
        reporter.fail(f"manifest.json missing in {branch_dir}")
        return reporter.exit_code

    rounds = _latest_round_per_agent(branch_dir)
    if "code-reviewer" not in rounds:
        reporter.fail("no round file for agent 'code-reviewer'")
    if "math-auditor" not in rounds:
        reporter.fail("no round file for agent 'math-auditor'")
    if reporter.exit_code >= EXIT_FAIL:
        return reporter.exit_code

    cr = rounds["code-reviewer"]
    ma = rounds["math-auditor"]

    if cr.get("verdict") == "approve":
        reporter.ok("code-reviewer latest round verdict=approve")
    else:
        reporter.fail(f"code-reviewer latest round verdict={cr.get('verdict')!r}, expected approve")

    summary = ma.get("contract_only_summary") or ""
    if summary.strip():
        reporter.ok("math-auditor produced contract_only_summary (contract-first audit happened)")
    else:
        reporter.fail("math-auditor missing or empty contract_only_summary; contract-first guarantee broken")

    # Aggregate findings across all round files, deduped by id, taking the latest
    # ack overall. Order rounds by (round_number, agent) so that later rounds win.
    all_round_files = sorted(branch_dir.glob("round-*.json"))
    seen: dict[str, dict] = {}
    for fp in all_round_files:
        try:
            doc = load_json_file(fp)
        except json.JSONDecodeError:
            continue
        for finding in doc.get("findings", []) or []:
            fid = finding.get("id")
            if not fid:
                continue
            seen[fid] = finding  # later rounds overwrite earlier

    unresolved = 0
    for fid, finding in seen.items():
        sev = finding.get("severity")
        if sev not in ("blocker", "major"):
            continue
        ack = finding.get("ack") or {}
        stance = ack.get("stance")
        if stance == "resolved":
            continue
        reporter.fail(
            f"finding {fid} severity={sev} not resolved (latest ack stance={stance!r})"
        )
        unresolved += 1
    if unresolved == 0:
        reporter.ok(f"all blocker/major findings ({sum(1 for f in seen.values() if f.get('severity') in ('blocker','major'))}) resolved")
    return reporter.exit_code

def cmd_all(args: argparse.Namespace) -> int:
    """Run every standalone check in sequence; review-gate is opt-in via --branch."""
    repo_root = Path(args.repo_root).resolve()
    profile_path = repo_root / ".banking-profile"
    if not profile_path.is_file():
        print("[CFG]  .banking-profile absent; banking gates do not apply to this repo.")
        return EXIT_OK

    aggregated = EXIT_OK
    sub_args = argparse.Namespace(repo_root=str(repo_root), profile=None, branch=getattr(args, "branch", None))

    print("== profile-validate ==")
    aggregated = max(aggregated, cmd_profile_validate(sub_args))
    print("== contract-validate ==")
    aggregated = max(aggregated, cmd_contract_validate(sub_args))
    print("== contract-presence ==")
    aggregated = max(aggregated, cmd_contract_presence(sub_args))
    print("== floats-check ==")
    aggregated = max(aggregated, cmd_floats_check(sub_args))

    if sub_args.branch:
        print(f"== review-validate {sub_args.branch} ==")
        aggregated = max(aggregated, cmd_review_validate(sub_args))
        print(f"== review-gate {sub_args.branch} ==")
        aggregated = max(aggregated, cmd_review_gate(sub_args))
    else:
        print("[WARN] --branch not provided; review-validate and review-gate skipped")

    print(f"== bankcheck exit code: {aggregated} ==")
    return aggregated
